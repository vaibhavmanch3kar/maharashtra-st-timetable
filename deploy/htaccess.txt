# =====================================================================
# MSRTC Timetable — Apache configuration
# Use only if the host runs Apache instead of nginx. Requires
# AllowOverride All on the directory, plus mod_rewrite, mod_headers,
# mod_deflate and mod_expires enabled.
# =====================================================================

DirectoryIndex index.html
Options -Indexes -MultiViews

<IfModule mod_mime.c>
  AddType application/manifest+json .webmanifest
  AddType application/json          .json
  AddType text/javascript           .js
  AddType font/woff2                .woff2
  AddType image/webp                .webp
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript \
    application/javascript application/json application/manifest+json image/svg+xml
</IfModule>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options   "nosniff"
  Header always set X-Frame-Options          "SAMEORIGIN"
  Header always set Referrer-Policy          "strict-origin-when-cross-origin"
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains" "expr=%{HTTPS} == 'on'"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval' https://cdn.tailwindcss.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data:; connect-src 'self'; frame-ancestors 'self'; base-uri 'self'; form-action 'self'"
  Header unset X-Powered-By

  # A cached service worker can never be replaced. Never cache it.
  <FilesMatch "^sw\.js$">
    Header always set Cache-Control "no-cache, no-store, must-revalidate"
    Header always set Service-Worker-Allowed "/"
  </FilesMatch>

  # The timetable lives inside index.html, so index.html must revalidate.
  <FilesMatch "^index\.html$">
    Header always set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>

  <FilesMatch "\.(png|jpg|jpeg|webp|svg|ico|woff2?)$">
    Header always set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
</IfModule>

<IfModule mod_rewrite.c>
  RewriteEngine On

  # Service workers do not register over plain http. Force TLS.
  RewriteCond %{HTTPS} off
  RewriteCond %{HTTP_HOST} !^localhost [NC]
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

  # sw.js and the manifest must resolve to real files, never the fallback.
  RewriteRule ^(sw\.js|manifest\.json)$ - [L]

  # Deep links fall back to the app shell.
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^ index.html [L]
</IfModule>

<FilesMatch "^\.">
  Require all denied
</FilesMatch>
